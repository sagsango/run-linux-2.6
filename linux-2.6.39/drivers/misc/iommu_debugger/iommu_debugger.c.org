/*
 * iommu_debugger.c
 *
 * Linux 2.6.39
 *
 * Controlled IOMMU/DMA mapping experiment.
 *
 * debugfs:
 *
 *   /sys/kernel/debug/iommu_debugger/control
 *
 * Commands:
 *
 *   echo 1 > control    Allocate buffer + DMA map
 *   echo 2 > control    DMA unmap + free buffer
 *   echo 3 > control    Dump current mapping
 *
 * Module parameter:
 *
 *   bdf=0000:00:03.0
 *
 * Example:
 *
 *   insmod iommu_debugger.ko bdf=0000:00:03.0
 */

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/pci.h>
#include <linux/dma-mapping.h>
#include <linux/slab.h>
#include <linux/debugfs.h>
#include <linux/fs.h>
#include <linux/uaccess.h>
#include <linux/string.h>
#include <linux/io.h>

#define DRIVER_NAME             "iommu_debugger"
#define DMA_TEST_SIZE           PAGE_SIZE

#define CMD_MAP                 1
#define CMD_UNMAP               2
#define CMD_DUMP                3

static struct dentry *debugfs_dir;
static struct dentry *debugfs_control;

/*
 * User-selected PCI BDF.
 *
 * Example:
 *
 * insmod iommu_debugger.ko bdf=0000:00:03.0
 */
static char *bdf = "0000:00:03.0";
module_param(bdf, charp, 0444);
MODULE_PARM_DESC(bdf, "PCI BDF, e.g. 0000:00:03.0");

/*
 * PCI device used for DMA mapping.
 */
static struct pci_dev *test_pdev;

/*
 * DMA buffer.
 */
static void *dma_buf;

/*
 * DMA address returned by dma_map_single().
 *
 * IMPORTANT:
 *
 * This is the address the DEVICE should use.
 *
 * It is not necessarily the physical address.
 */
static dma_addr_t dma_handle;

static size_t dma_size = DMA_TEST_SIZE;

static int dma_mapped;


/*
 * ------------------------------------------------------------
 * Dump mapping information
 * ------------------------------------------------------------
 */
static void iommu_debug_dump(void)
{
	unsigned long phys;

	printk(KERN_INFO
	       "IOMMU-DBG: ========================================\n");

	printk(KERN_INFO
	       "IOMMU-DBG: DMA TEST STATE\n");

	if (!test_pdev) {
		printk(KERN_INFO
		       "IOMMU-DBG: PCI device: NONE\n");
		goto out;
	}

	printk(KERN_INFO
	       "IOMMU-DBG: PCI device : %04x:%02x:%02x.%d\n",
	       pci_domain_nr(test_pdev->bus),
	       test_pdev->bus->number,
	       PCI_SLOT(test_pdev->devfn),
	       PCI_FUNC(test_pdev->devfn));

	printk(KERN_INFO
	       "IOMMU-DBG: vendor     : 0x%04x\n",
	       test_pdev->vendor);

	printk(KERN_INFO
	       "IOMMU-DBG: device     : 0x%04x\n",
	       test_pdev->device);

	printk(KERN_INFO
	       "IOMMU-DBG: DMA mapped : %s\n",
	       dma_mapped ? "YES" : "NO");

	if (!dma_buf)
		goto out;

	phys = virt_to_phys(dma_buf);

	printk(KERN_INFO
	       "IOMMU-DBG: CPU VA     : %p\n",
	       dma_buf);

	printk(KERN_INFO
	       "IOMMU-DBG: PHYS       : 0x%lx\n",
	       phys);

	printk(KERN_INFO
	       "IOMMU-DBG: DMA/IOVA   : 0x%llx\n",
	       (unsigned long long)dma_handle);

	printk(KERN_INFO
	       "IOMMU-DBG: SIZE       : %lu\n",
	       (unsigned long)dma_size);

	printk(KERN_INFO
	       "IOMMU-DBG: direction  : DMA_TO_DEVICE\n");

out:
	printk(KERN_INFO
	       "IOMMU-DBG: ========================================\n");
}


/*
 * ------------------------------------------------------------
 * Allocate + map
 * ------------------------------------------------------------
 */
static int iommu_debug_map(void)
{
	unsigned long phys;

	if (dma_mapped) {
		printk(KERN_INFO
		       "IOMMU-DBG: buffer already mapped\n");
		return 0;
	}

	if (!test_pdev) {
		printk(KERN_ERR
		       "IOMMU-DBG: no PCI device selected\n");
		return -ENODEV;
	}

	/*
	 * Allocate one physically contiguous page.
	 *
	 * kmalloc() memory is appropriate for this simple experiment.
	 */
	dma_buf = kmalloc(DMA_TEST_SIZE, GFP_KERNEL);

	if (!dma_buf) {
		printk(KERN_ERR
		       "IOMMU-DBG: kmalloc failed\n");
		return -ENOMEM;
	}

	/*
	 * Put a recognizable pattern into the buffer.
	 */
	memset(dma_buf, 0xaa, DMA_TEST_SIZE);

	phys = virt_to_phys(dma_buf);

	printk(KERN_INFO
	       "IOMMU-DBG: ----------------------------------------\n");

	printk(KERN_INFO
	       "IOMMU-DBG: starting DMA MAP\n");

	printk(KERN_INFO
	       "IOMMU-DBG: device     = %04x:%02x:%02x.%d\n",
	       pci_domain_nr(test_pdev->bus),
	       test_pdev->bus->number,
	       PCI_SLOT(test_pdev->devfn),
	       PCI_FUNC(test_pdev->devfn));

	printk(KERN_INFO
	       "IOMMU-DBG: CPU VA     = %p\n",
	       dma_buf);

	printk(KERN_INFO
	       "IOMMU-DBG: physical   = 0x%lx\n",
	       phys);

	printk(KERN_INFO
	       "IOMMU-DBG: size       = %lu\n",
	       (unsigned long)DMA_TEST_SIZE);

	/*
	 * THIS is the important operation.
	 *
	 * Linux's DMA layer will call the appropriate architecture/IOMMU
	 * implementation.
	 *
	 * On an Intel IOMMU-enabled system this can result in an IOVA
	 * being allocated and an I/O page-table mapping being installed.
	 */
	dma_handle = dma_map_single(&test_pdev->dev,
				    dma_buf,
				    DMA_TEST_SIZE,
				    DMA_TO_DEVICE);

	if (dma_mapping_error(&test_pdev->dev, dma_handle)) {
		printk(KERN_ERR
		       "IOMMU-DBG: dma_map_single() FAILED\n");

		kfree(dma_buf);
		dma_buf = NULL;

		return -EIO;
	}

	dma_mapped = 1;

	printk(KERN_INFO
	       "IOMMU-DBG: dma_map_single() SUCCESS\n");

	printk(KERN_INFO
	       "IOMMU-DBG: DMA/IOVA  = 0x%llx\n",
	       (unsigned long long)dma_handle);

	printk(KERN_INFO
	       "IOMMU-DBG: ----------------------------------------\n");

	return 0;
}


/*
 * ------------------------------------------------------------
 * Unmap + free
 * ------------------------------------------------------------
 */
static int iommu_debug_unmap(void)
{
	if (!dma_mapped) {
		printk(KERN_INFO
		       "IOMMU-DBG: no DMA mapping exists\n");
		return 0;
	}

	printk(KERN_INFO
	       "IOMMU-DBG: ----------------------------------------\n");

	printk(KERN_INFO
	       "IOMMU-DBG: starting DMA UNMAP\n");

	printk(KERN_INFO
	       "IOMMU-DBG: DMA/IOVA = 0x%llx\n",
	       (unsigned long long)dma_handle);

	dma_unmap_single(&test_pdev->dev,
			 dma_handle,
			 DMA_TEST_SIZE,
			 DMA_TO_DEVICE);

	printk(KERN_INFO
	       "IOMMU-DBG: dma_unmap_single() COMPLETE\n");

	dma_mapped = 0;
	dma_handle = 0;

	kfree(dma_buf);
	dma_buf = NULL;

	printk(KERN_INFO
	       "IOMMU-DBG: buffer freed\n");

	printk(KERN_INFO
	       "IOMMU-DBG: ----------------------------------------\n");

	return 0;
}


/*
 * ------------------------------------------------------------
 * debugfs write
 * ------------------------------------------------------------
 *
 * echo 1 > control
 * echo 2 > control
 * echo 3 > control
 */
static ssize_t iommu_debug_write(struct file *file,
				 const char __user *buf,
				 size_t count,
				 loff_t *ppos)
{
	char command[16];
	unsigned long cmd;

	if (count >= sizeof(command))
		return -EINVAL;

	if (copy_from_user(command, buf, count))
		return -EFAULT;

	command[count] = '\0';

	cmd = simple_strtoul(command, NULL, 10);

	switch (cmd) {

	case CMD_MAP:
		iommu_debug_map();
		break;

	case CMD_UNMAP:
		iommu_debug_unmap();
		break;

	case CMD_DUMP:
		iommu_debug_dump();
		break;

	default:
		printk(KERN_INFO
		       "IOMMU-DBG: unknown command %lu\n",
		       cmd);

		printk(KERN_INFO
		       "IOMMU-DBG: commands:\n");

		printk(KERN_INFO
		       "IOMMU-DBG:   1 = MAP\n");

		printk(KERN_INFO
		       "IOMMU-DBG:   2 = UNMAP\n");

		printk(KERN_INFO
		       "IOMMU-DBG:   3 = DUMP\n");

		return -EINVAL;
	}

	return count;
}


/*
 * ------------------------------------------------------------
 * debugfs read
 * ------------------------------------------------------------
 */
static ssize_t iommu_debug_read(struct file *file,
				char __user *buf,
				size_t count,
				loff_t *ppos)
{
	char msg[256];
	int len;

	if (*ppos != 0)
		return 0;

	len = snprintf(msg, sizeof(msg),
		       "IOMMU debugger\n"
		       "\n"
		       "PCI device : %s\n"
		       "mapped     : %s\n"
		       "\n"
		       "Commands:\n"
		       "  echo 1 > control    MAP\n"
		       "  echo 2 > control    UNMAP\n"
		       "  echo 3 > control    DUMP\n",
		       bdf,
		       dma_mapped ? "yes" : "no");

	if (copy_to_user(buf, msg, len))
		return -EFAULT;

	*ppos += len;

	return len;
}


static const struct file_operations iommu_debug_fops = {
	.owner	= THIS_MODULE,
	.read	= iommu_debug_read,
	.write	= iommu_debug_write,
};


/*
 * ------------------------------------------------------------
 * Find PCI device
 * ------------------------------------------------------------
 */
#if 1
static int iommu_debug_find_pci_device(void)
{
	unsigned int domain;
	unsigned int bus;
	unsigned int slot;
	unsigned int function;

	if (sscanf(bdf, "%x:%x:%x.%x",
		   &domain,
		   &bus,
		   &slot,
		   &function) != 4) {

		printk(KERN_ERR
		       "IOMMU-DBG: invalid BDF: %s\n", bdf);

		return -EINVAL;
	}

	test_pdev = pci_get_domain_bus_and_slot(domain,
						bus,
						PCI_DEVFN(slot, function));

	if (!test_pdev) {
		printk(KERN_ERR
		       "IOMMU-DBG: PCI device %s not found\n", bdf);

		return -ENODEV;
	}

	printk(KERN_INFO
	       "IOMMU-DBG: selected PCI device %s\n", bdf);

	printk(KERN_INFO
	       "IOMMU-DBG: vendor = 0x%04x device = 0x%04x\n",
	       test_pdev->vendor,
	       test_pdev->device);

	return 0;
}
#else
static int iommu_debug_find_pci_device(void)
{
	unsigned int domain;
	unsigned int bus;
	unsigned int devfn;
	int ret;

	/*
	 * Parse:
	 *
	 *   DDDD:BB:DD.F
	 */
	ret = sscanf(bdf,
		     "%x:%x:%x.%x",
		     &domain,
		     &bus,
		     &devfn,
		     &ret);

	/*
	 * sscanf() above is awkward for PCI BDF parsing because
	 * devfn needs slot/function conversion.
	 *
	 * Instead use pci_get_domain_bus_and_slot() below after
	 * parsing manually.
	 */

	if (ret != 4) {
		printk(KERN_ERR
		       "IOMMU-DBG: invalid BDF: %s\n",
		       bdf);

		return -EINVAL;
	}

	/*
	 * The third parsed value is the PCI slot and the fourth is
	 * the function. Re-parse with explicit variables.
	 */
	{
		unsigned int slot;
		unsigned int function;
		struct pci_dev *pdev;

		if (sscanf(bdf,
			   "%x:%x:%x.%x",
			   &domain,
			   &bus,
			   &slot,
			   &function) != 4) {

			printk(KERN_ERR
			       "IOMMU-DBG: invalid BDF: %s\n",
			       bdf);

			return -EINVAL;
		}

		devfn = PCI_DEVFN(slot, function);

		pdev = pci_get_domain_bus_and_slot(domain,
						   bus,
						   devfn);

		if (!pdev) {
			printk(KERN_ERR
			       "IOMMU-DBG: PCI device %s not found\n",
			       bdf);

			return -ENODEV;
		}

		test_pdev = pdev;
	}

	printk(KERN_INFO
	       "IOMMU-DBG: selected PCI device %s\n",
	       bdf);

	printk(KERN_INFO
	       "IOMMU-DBG: vendor = 0x%04x device = 0x%04x\n",
	       test_pdev->vendor,
	       test_pdev->device);

	return 0;
}
#endif

/*
 * ------------------------------------------------------------
 * Module initialization
 * ------------------------------------------------------------
 */
static int __init iommu_debugger_init(void)
{
	int ret;

	printk(KERN_INFO
	       "IOMMU-DBG: initializing\n");

	ret = iommu_debug_find_pci_device();

	if (ret)
		return ret;

	/*
	 * Create:
	 *
	 * /sys/kernel/debug/iommu_debugger
	 */
	debugfs_dir = debugfs_create_dir(DRIVER_NAME, NULL);

	if (!debugfs_dir) {
		printk(KERN_ERR
		       "IOMMU-DBG: debugfs directory creation failed\n");

		pci_dev_put(test_pdev);
		test_pdev = NULL;

		return -ENOMEM;
	}

	/*
	 * Create:
	 *
	 * /sys/kernel/debug/iommu_debugger/control
	 */
	debugfs_control = debugfs_create_file("control",
					      0644,
					      debugfs_dir,
					      NULL,
					      &iommu_debug_fops);

	if (!debugfs_control) {
		printk(KERN_ERR
		       "IOMMU-DBG: debugfs file creation failed\n");

		debugfs_remove(debugfs_dir);

		pci_dev_put(test_pdev);
		test_pdev = NULL;

		return -ENOMEM;
	}

	printk(KERN_INFO
	       "IOMMU-DBG: debugfs ready\n");

	printk(KERN_INFO
	       "IOMMU-DBG: control = "
	       "/sys/kernel/debug/iommu_debugger/control\n");

	printk(KERN_INFO
	       "IOMMU-DBG: echo 1 > control  -> MAP\n");

	printk(KERN_INFO
	       "IOMMU-DBG: echo 2 > control  -> UNMAP\n");

	printk(KERN_INFO
	       "IOMMU-DBG: echo 3 > control  -> DUMP\n");

	return 0;
}


/*
 * ------------------------------------------------------------
 * Module cleanup
 * ------------------------------------------------------------
 */
static void __exit iommu_debugger_exit(void)
{
	printk(KERN_INFO
	       "IOMMU-DBG: shutting down\n");

	/*
	 * Never leave a DMA mapping active.
	 */
	if (dma_mapped)
		iommu_debug_unmap();

	debugfs_remove(debugfs_control);
	debugfs_remove(debugfs_dir);

	if (test_pdev) {
		pci_dev_put(test_pdev);
		test_pdev = NULL;
	}

	printk(KERN_INFO
	       "IOMMU-DBG: unloaded\n");
}


module_init(iommu_debugger_init);
module_exit(iommu_debugger_exit);
